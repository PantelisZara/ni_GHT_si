cmake_minimum_required(VERSION 3.20)
project(ni_GHT_si VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The metrics engine currently supports Linux only")
endif()
find_package(Threads REQUIRED)
option(NIGHT_SANITIZERS "Enable address/undefined behavior sanitizers" OFF)
add_library(night_metrics src/common.cpp src/system.cpp src/process.cpp src/io.cpp src/engine.cpp src/application_scope.cpp
    src/descriptor_budget.cpp)
target_include_directories(night_metrics PUBLIC include)
target_compile_features(night_metrics PUBLIC cxx_std_20)
set_target_properties(night_metrics PROPERTIES CXX_EXTENSIONS OFF)
target_compile_options(night_metrics PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(night_metrics PUBLIC Threads::Threads)
add_library(night_process_handles src/pinned_process.cpp src/process_control.cpp src/application_control.cpp)
target_link_libraries(night_process_handles PUBLIC night_metrics)
target_compile_options(night_process_handles PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(NIGHT_SANITIZERS)
    target_compile_options(night_metrics PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(night_metrics PUBLIC -fsanitize=address,undefined)
endif()
add_executable(night-metrics src/main.cpp)
target_link_libraries(night-metrics PRIVATE night_metrics)
include(CTest)
option(NIGHT_NVML "Build the optional NVIDIA NVML collector using an installed SDK" OFF)
if(NIGHT_NVML)
    find_path(NVML_INCLUDE_DIR nvml.h HINTS /opt/cuda/targets/x86_64-linux/include /opt/cuda/include REQUIRED)
    add_library(night_gpu STATIC src/gpu.cpp)
    target_include_directories(night_gpu PRIVATE "${NVML_INCLUDE_DIR}")
    target_link_libraries(night_gpu PUBLIC night_metrics PRIVATE ${CMAKE_DL_LIBS})
    target_compile_options(night_gpu PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(BUILD_TESTING AND CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang)$")
        add_executable(gpu-tests tests/gpu_tests.cpp)
        target_include_directories(gpu-tests PRIVATE "${NVML_INCLUDE_DIR}")
        target_link_libraries(gpu-tests PRIVATE night_gpu)
        target_link_options(gpu-tests PRIVATE "LINKER:--wrap=dlopen" "LINKER:--wrap=dlsym" "LINKER:--wrap=dlclose")
        target_compile_options(gpu-tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
        add_test(NAME gpu-contracts COMMAND gpu-tests)
        set_tests_properties(gpu-contracts PROPERTIES TIMEOUT 10)
    endif()
endif()
